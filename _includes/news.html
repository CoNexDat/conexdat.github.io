{%- comment -%} Renders the news collection as a timeline, filtered by language. Required: include.lang (es|en|fr), optional include.limit (default 10). Each item in _news/ has front matter `date`, `lang`, `title`, `link` and optional `month_only: true` (badge shows month and year, no day); the body is one short sentence. Dates are written in the page language. {%- endcomment -%} {%- assign lang = include.lang | default: site.default_lang -%} {%- assign limit = include.limit | default: 10 -%} {%- assign news_items = site.news | where: "lang", lang | sort: "date" | reverse -%} {%- if lang == "es" -%} {%- assign months = "ene,feb,mar,abr,may,jun,jul,ago,sep,oct,nov,dic" | split: "," -%} {%- elsif lang == "fr" -%} {%- assign months = "janv.,févr.,mars,avr.,mai,juin,juil.,août,sept.,oct.,nov.,déc." | split: "," -%} {%- else -%} {%- assign months = "Jan,Feb,Mar,Apr,May,Jun,Jul,Aug,Sep,Oct,Nov,Dec" | split: "," -%} {%- endif -%} {% if news_items.size > 0 %}
    {% for item in news_items limit: limit %} {%- assign m = item.date | date: "%-m" | minus: 1 -%} {%- assign y = item.date | date: "%Y" -%}
  1. {%- if item.month_only -%} {{ months[m] }} {{ y }} {%- else -%} {{ item.date | date: "%-d" }} {{ months[m] }} {{ y }} {%- endif -%}
    {%- if item.link and item.link != "" -%} {{ item.title }} {%- else -%} {{ item.title }} {%- endif -%}
    {{ item.content | markdownify }}
    2. {% endfor %}
{% endif %}
